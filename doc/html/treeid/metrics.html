treeid.metrics

Segmentation metrics for thin branch structures.

Pixel-overlap scores (IoU, Dice) under-reward thin structures because a one-pixel offset along a branch only a few pixels wide destroys most of the overlap. The module therefore also provides tolerance-based scores that ask the question an orchard robot actually cares about: is there predicted branch within tol pixels of every true branch, and is every predicted branch within tol pixels of a true branch?

All functions take boolean arrays of identical shape and return Python floats. Empty-set conventions are documented per function so that results are never nan silently.

 22def _check(pred: np.ndarray, gt: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
 23    """Validate and cast a prediction / ground-truth pair to bool.
 24
 25    Raises:
 26        ValueError: If the shapes differ.
 27    """
 28    pred = np.asarray(pred).astype(bool)
 29    gt = np.asarray(gt).astype(bool)
 30    if pred.shape != gt.shape:
 31        raise ValueError(f"shape mismatch: pred {pred.shape} vs gt {gt.shape}")
 32    return pred, gt

 35def iou(pred: np.ndarray, gt: np.ndarray) -> float:
 36    """Intersection over union (Jaccard index).
 37
 38    Returns 1.0 when both masks are empty (nothing to find, nothing predicted).
 39    """
 40    pred, gt = _check(pred, gt)
 41    union = np.logical_or(pred, gt).sum()
 42    if union == 0:
 43        return 1.0
 44    return float(np.logical_and(pred, gt).sum() / union)

 47def dice(pred: np.ndarray, gt: np.ndarray) -> float:
 48    """Dice / F1 overlap, ``2|A∩B| / (|A| + |B|)``. 1.0 when both are empty."""
 49    pred, gt = _check(pred, gt)
 50    s = pred.sum() + gt.sum()
 51    if s == 0:
 52        return 1.0
 53    return float(2 * np.logical_and(pred, gt).sum() / s)

 56def _dist_to(mask: np.ndarray) -> np.ndarray:
 57    """Euclidean distance (pixels) from every pixel to the nearest ``True`` of ``mask``.
 58
 59    If ``mask`` is empty every distance is ``inf``.
 60    """
 61    if not mask.any():
 62        return np.full(mask.shape, np.inf)
 63    return ndimage.distance_transform_edt(~mask)

 66def tolerant_precision_recall(
 67    pred: np.ndarray, gt: np.ndarray, tol: float = 2.0, skeleton: bool = True
 68) -> tuple[float, float, float]:
 69    """Precision, recall and F1 with a spatial tolerance.
 70
 71    A true branch pixel counts as *recalled* if any predicted pixel lies within
 72    ``tol`` pixels of it; a predicted pixel counts as *precise* if any true pixel
 73    lies within ``tol``. With ``skeleton=True`` both masks are first reduced to
 74    one-pixel-wide centrelines so that thick and thin branches are weighted by
 75    length rather than area, which is the quantity a pruning robot needs.
 76
 77    Args:
 78        pred: Predicted branch mask.
 79        gt: Ground-truth branch mask.
 80        tol: Matching tolerance in pixels (>= 0).
 81        skeleton: Compare centrelines instead of full masks.
 82
 83    Returns:
 84        ``(precision, recall, f1)``. Precision is 1.0 when nothing is predicted;
 85        recall is 1.0 when the ground truth is empty.
 86
 87    Raises:
 88        ValueError: If ``tol`` is negative or shapes differ.
 89    """
 90    if tol < 0:
 91        raise ValueError("tol must be non-negative")
 92    pred, gt = _check(pred, gt)
 93    if skeleton:
 94        pred = skeletonize(pred) if pred.any() else pred
 95        gt = skeletonize(gt) if gt.any() else gt
 96    d_to_gt = _dist_to(gt)
 97    d_to_pred = _dist_to(pred)
 98    precision = float((d_to_gt[pred] <= tol).mean()) if pred.any() else 1.0
 99    recall = float((d_to_pred[gt] <= tol).mean()) if gt.any() else 1.0
100    f1 = 0.0 if precision + recall == 0 else 2 * precision * recall / (precision + recall)
101    return precision, recall, f1

104def boundary_f1(pred: np.ndarray, gt: np.ndarray, tol: float = 2.0) -> float:
105    """Boundary F1 score (Csurka et al., 2013) used by Chen et al. (2021).
106
107    Boundaries are the mask pixels removed by a 3x3 erosion. The score is the
108    tolerant F1 of the two boundary maps (no skeletonisation).
109    """
110    pred, gt = _check(pred, gt)
111    pb = pred & ~ndimage.binary_erosion(pred)
112    gb = gt & ~ndimage.binary_erosion(gt)
113    return tolerant_precision_recall(pb, gb, tol=tol, skeleton=False)[2]

116def occluded_recall(
117    pred: np.ndarray, gt: np.ndarray, occluded: np.ndarray, tol: float = 2.0
118) -> float:
119    """Recall restricted to ground-truth centreline pixels hidden by foliage.
120
121    Args:
122        pred: Predicted branch mask.
123        gt: Ground-truth branch mask.
124        occluded: Boolean map of pixels where the branch is not visible in the
125            input image (e.g. leaf-covered). See
126            :func:`treeid.preprocess.foliage_mask`.
127        tol: Matching tolerance in pixels.
128
129    Returns:
130        Fraction of occluded centreline pixels recovered, or ``nan`` when the tree
131        has no occluded branch pixels (callers should use ``np.nanmean``).
132    """
133    pred, gt = _check(pred, gt)
134    occ = np.asarray(occluded, bool)
135    g = skeletonize(gt) if gt.any() else gt
136    g_occ = g & occ
137    if not g_occ.any():
138        return float("nan")
139    return float((_dist_to(pred)[g_occ] <= tol).mean())

142def shift_compensated_f1(pred: np.ndarray, gt: np.ndarray, tol: float = 4.0,
143                         max_shift: int = 24, step: int = 1) -> tuple[float, int]:
144    """Centreline F1 after the best horizontal re-registration of the label.
145
146    The ground truth is translated horizontally by every integer offset in
147    ``[-max_shift, max_shift]`` and the best tolerant F1 is kept. This removes the
148    effect of the label mis-registration found in the data audit and therefore
149    measures whether the *shape* of the predicted structure is right. It is an
150    optimistic score and is always reported next to the strict score.
151
152    Returns:
153        ``(best_f1, best_shift_px)``; when several shifts tie, the median of the
154        tied shifts is returned.
155    """
156    pred, gt = _check(pred, gt)
157    ps = skeletonize(pred) if pred.any() else pred
158    gs = skeletonize(gt) if gt.any() else gt
159    d_to_pred = _dist_to(ps)
160    scores = {}
161    w = gt.shape[1]
162    for k in range(-max_shift, max_shift + 1, step):
163        g = np.zeros_like(gs)
164        if k >= 0:
165            g[:, k:] = gs[:, : w - k]
166        else:
167            g[:, :k] = gs[:, -k:]
168        if not g.any() or not ps.any():
169            f = 1.0 if (not g.any() and not ps.any()) else 0.0
170        else:
171            prec = float((_dist_to(g)[ps] <= tol).mean())
172            rec = float((d_to_pred[g] <= tol).mean())
173            f = 0.0 if prec + rec == 0 else 2 * prec * rec / (prec + rec)
174        scores[k] = f
175    best = max(scores.values())
176    # Several shifts within the tolerance can tie; report the centre of the plateau.
177    ties = [k for k, f in scores.items() if f >= best - 1e-9]
178    return best, int(np.median(ties))

181def score_all(pred: np.ndarray, gt: np.ndarray, occluded: np.ndarray | None = None,
182              tol: float = 2.0) -> dict[str, float]:
183    """Compute every metric used in the report for one image.
184
185    Returns:
186        Dict with keys ``iou, dice, bf1, cl_precision, cl_recall, cl_f1`` and,
187        if ``occluded`` is given, ``occ_recall``.
188    """
189    p, r, f = tolerant_precision_recall(pred, gt, tol=tol, skeleton=True)
190    out = {
191        "iou": iou(pred, gt),
192        "dice": dice(pred, gt),
193        "bf1": boundary_f1(pred, gt, tol=tol),
194        "cl_precision": p,
195        "cl_recall": r,
196        "cl_f1": f,
197    }
198    if occluded is not None:
199        out["occ_recall"] = occluded_recall(pred, gt, occluded, tol=tol)
200    return out
